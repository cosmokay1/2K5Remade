cmake_minimum_required(VERSION 3.20)
project(RuntimePathsPosixCheck LANGUAGES C)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "Runtime path checks require a case-sensitive Linux filesystem")
endif()
add_executable(runtime_paths_check check.c ../../external/xboxrecomp/src/kernel/kernel_path.c)
target_include_directories(runtime_paths_check PRIVATE ../../external/xboxrecomp/src)
target_compile_definitions(runtime_paths_check PRIVATE _GNU_SOURCE)
target_compile_options(runtime_paths_check PRIVATE -UNDEBUG)
set_target_properties(runtime_paths_check PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED ON)
enable_testing()
add_test(NAME runtime_paths COMMAND runtime_paths_check)
set_tests_properties(runtime_paths PROPERTIES TIMEOUT 15)
