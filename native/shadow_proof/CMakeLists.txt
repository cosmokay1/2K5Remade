cmake_minimum_required(VERSION 3.20)
project(ShadowProof LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
enable_testing()
add_executable(shadow_reference reference_check.cpp)
add_test(NAME shadow_cpu_reference COMMAND shadow_reference "${CMAKE_CURRENT_BINARY_DIR}/reference")
if(WIN32)
  add_executable(shadow_capture_check capture_check.c)
  set_target_properties(shadow_capture_check PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED ON)
  target_compile_definitions(shadow_capture_check PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX _CRT_SECURE_NO_WARNINGS)
  target_link_libraries(shadow_capture_check PRIVATE d3d11)
  if(MSVC)
    target_compile_options(shadow_capture_check PRIVATE /UNDEBUG)
  else()
    target_compile_options(shadow_capture_check PRIVATE -UNDEBUG)
  endif()
  add_test(NAME shadow_renderer_capture COMMAND shadow_capture_check)
  find_program(DXC_EXECUTABLE NAMES dxc REQUIRED DOC "DirectX Shader Compiler with Shader Model 6.5 support")
  add_custom_command(OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/shadow.cso"
    COMMAND "${DXC_EXECUTABLE}" -T cs_6_5 -E main -WX -Fo "${CMAKE_CURRENT_BINARY_DIR}/shadow.cso" "${CMAKE_CURRENT_SOURCE_DIR}/shadow.hlsl"
    DEPENDS shadow.hlsl VERBATIM)
  add_custom_target(shadow_shader DEPENDS "${CMAKE_CURRENT_BINARY_DIR}/shadow.cso")
  add_executable(shadow_dxr dxr.cpp)
  target_compile_definitions(shadow_dxr PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
  target_link_libraries(shadow_dxr PRIVATE d3d12 dxgi)
  add_dependencies(shadow_dxr shadow_shader)
  add_test(NAME shadow_gpu_reference COMMAND shadow_dxr "${CMAKE_CURRENT_BINARY_DIR}/shadow.cso" "${CMAKE_CURRENT_BINARY_DIR}/gpu")
  add_test(NAME shadow_gpu_box_reference COMMAND shadow_dxr "${CMAKE_CURRENT_BINARY_DIR}/shadow.cso" "${CMAKE_CURRENT_BINARY_DIR}/gpu-box" --box)
  string(RANDOM LENGTH 12 ALPHABET 0123456789abcdef SHADOW_TEST_TOKEN)
  add_test(NAME shadow_gpu_live_reference COMMAND shadow_dxr "${CMAKE_CURRENT_BINARY_DIR}/shadow.cso" "${CMAKE_CURRENT_BINARY_DIR}/gpu-live" --live-check "Local\\shadow-proof-${SHADOW_TEST_TOKEN}" 3)
  set_tests_properties(shadow_gpu_reference shadow_gpu_box_reference shadow_gpu_live_reference PROPERTIES SKIP_RETURN_CODE 77 TIMEOUT 60)
endif()
