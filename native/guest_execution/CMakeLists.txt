cmake_minimum_required(VERSION 3.20)
project(GuestExecutionCheck LANGUAGES C)
add_subdirectory(../../external/xboxrecomp/src/platform platform)
add_executable(guest_execution_check check.c ../../external/xboxrecomp/src/kernel/xbox_ggl.c)
target_include_directories(guest_execution_check PRIVATE ../../external/xboxrecomp/src/kernel)
target_link_libraries(guest_execution_check PRIVATE platform)
set_target_properties(guest_execution_check PROPERTIES C_STANDARD 11)
if(WIN32)
  target_link_libraries(guest_execution_check PRIVATE dbghelp)
endif()
enable_testing()
add_test(NAME guest_execution_owned COMMAND guest_execution_check enabled)
set_tests_properties(guest_execution_owned PROPERTIES ENVIRONMENT "RECOMP_GGL=1;RECOMP_GGL_WATCHDOG=0" TIMEOUT 15)
add_test(NAME guest_execution_disabled COMMAND guest_execution_check disabled)
set_tests_properties(guest_execution_disabled PROPERTIES ENVIRONMENT "RECOMP_GGL=0;RECOMP_GGL_WATCHDOG=0" TIMEOUT 15)
